\section{Functionals and variational derivatives}

\begin{definition}[Functional]
  A \emph{functional} $\mathcal{F}[\phi]$ assigns a real number
  to each admissible field $\phi$.
\end{definition}

\begin{definition}[Variational derivative]
  The \emph{variational derivative} $\frac{\delta\mathcal{F}}{\delta\phi}$
  is defined by
  \[
    \mathcal{F}[\phi + \varepsilon\psi]
    = \mathcal{F}[\phi]
    + \varepsilon \int \frac{\delta\mathcal{F}}{\delta\phi}\,\psi\,dx
    + O(\varepsilon^2).
  \]
\end{definition}

\begin{example}[Gradient energy functional]
  For $\mathcal{F}[\phi] = \int \frac{1}{2}|\nabla\phi|^2\,dx$,
  \[
    \frac{\delta\mathcal{F}}{\delta\phi} = -\Delta\phi.
  \]
  Proof by integration by parts.
\end{example}

\section{The Euler--Lagrange equation}

\begin{theorem}[Euler--Lagrange]
  For $\mathcal{F}[\phi] = \int L(\phi, \nabla\phi)\,dx$,
  critical points satisfy
  \[
    \frac{\partial L}{\partial \phi}
    - \nabla \cdot \frac{\partial L}{\partial(\nabla\phi)}
    = 0.
  \]
\end{theorem}

\begin{DerivRoadmap}[Free-energy descent]
  \textbf{Goal:} Derive the Allen--Cahn equation from a
  variational principle.

  \textbf{Starting point:}
  $\mathcal{F}[\phi] = \int [\frac{\epsilon^2}{2}|\nabla\phi|^2
  + W(\phi)]\,dx$.

  \textbf{Method:}
  (1) Compute $\delta\mathcal{F}/\delta\phi$ using Euler--Lagrange.
  (2) Set $\partial_t\phi = -M\,\delta\mathcal{F}/\delta\phi$.
  (3) Verify $d\mathcal{F}/dt \leq 0$.

  \textbf{Result:} Allen--Cahn equation with energy dissipation.
\end{DerivRoadmap}
