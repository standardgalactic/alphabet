\chapter{Dynamic Scaling and Universality}
\label{ch:scaling-laws}

\section{The dynamic scaling hypothesis}

\begin{definition}[Dynamic scaling]
  A coarsening system obeys \emph{dynamic scaling} if
  its two-point correlation function satisfies
  \[
    C(r,t) = f\!\left(\frac{r}{\ell(t)}\right)
  \]
  for a universal scaling function $f$ and characteristic
  length $\ell(t)$, defined operationally by
  \[
    C(\ell(t),t) = \tfrac{1}{2}C(0,t).
  \]
\end{definition}

\begin{theorem}[Self-similar morphology]
  \label{thm:self-similar}
  If dynamic scaling holds, all morphological statistics
  at time $t$ depend on coordinates only through $r/\ell(t)$.
\end{theorem}

\begin{proof}
  Replace $r_i$ by $r_i/\ell(t)$ in any $n$-point function.
  Under dynamic scaling, explicit $t$-dependence cancels,
  leaving dependence only on dimensionless ratios.
\end{proof}

\section{Universality}

\begin{definition}[Universality class]
  Two systems belong to the same \emph{universality class}
  if they share the same exponents $z$, $\nu$, $\eta$, etc.\
  regardless of microscopic details.
\end{definition}

\begin{TechNote}[Universality principle]
  Microscopic differences are often irrelevant to macroscopic
  structure.
  Only symmetry, conservation laws, and dimensionality
  determine the universality class.
  This principle prepares the reader for RSVP's claim that
  many cosmological observables may be consequences of
  universality classes rather than unique signatures of a
  particular microscopic theory.
\end{TechNote}
