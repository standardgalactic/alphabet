\chapter{Entropy Production Theorems}
\label{ch:entropy-production}

\section{RSVP scaling regimes}

Let $\ell(t)$ be the characteristic coarsening length.
A general RSVP scaling form is
\[
  \frac{d\ell}{dt}
  = A\ell^{-1} + B\ell^{-2} + Cv - D\,Ro_{\mathrm{RSVP}}\,\ell^{-p}.
\]

\begin{definition}[Dimensionless RSVP parameters]
  \label{def:rsvp-params}
  \begin{align*}
    Pe &= \frac{v\ell}{\kappa}
    \quad\text{(Péclet: advection vs.\ diffusion)},\\
    \Gamma &= \frac{\lambda}{\nu\kappa}
    \quad\text{(coupling ratio)},\\
    Ro_{\mathrm{RSVP}} &= \frac{\eta|\nabla\times\mathbf{v}|^2}
      {\nu|\nabla\mathbf{v}|^2 + \varepsilon}
    \quad\text{(rotational transport number)}.
  \end{align*}
\end{definition}

\begin{proposition}[Dominant balance determines exponent]
  \label{prop:dom-balance}
  \statusproven{}
  If $\frac{d\ell}{dt} \sim C\ell^{-m}$, then
  $\ell(t) \sim t^{1/(m+1)}$ and $z = m+1$.
\end{proposition}

\begin{proof}
  $\ell^m\,d\ell = C\,dt$.
  Integrate: $\frac{1}{m+1}\ell^{m+1} = Ct + C_0$.
  For large $t$: $\ell^{m+1} \sim t$, so $\ell \sim t^{1/(m+1)}$.
  Since $\ell \sim t^{1/z}$, we obtain $z = m+1$.
\end{proof}

\begin{conjecture}[RSVP coarsening phase diagram]
  \label{conj:rsvp-phase-diagram}
  \statusconjectured{}
  \[
    z_{\mathrm{RSVP}} = z(Pe, \Gamma, Ro_{\mathrm{RSVP}}, \ldots).
  \]
  The phase diagram has regimes:
  $Pe \ll 1, Ro \ll 1$: Allen--Cahn dominated, $z \approx 2$;
  $Pe \ll 1, Ro \ll 1$, conserved $L$: Cahn--Hilliard, $z \approx 3$;
  $Pe \gg 1$: advection-accelerated, $z < 2$;
  $Ro \gg 1$: vorticity-stabilised, $z > 3$.
\end{conjecture}

\begin{OpenQuestion}[RSVP phase diagram]
  Derive $z(Pe, \Gamma, Ro_{\mathrm{RSVP}})$ analytically or
  by direct numerical simulation of the coupled RSVP system.
\end{OpenQuestion}
