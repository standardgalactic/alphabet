\section{RSVP as sections of the admissibility sheaf}

The RSVP state $(\Phi, \mathbf{v}, S, \ell)$ is not merely
a collection of fields.
It is a \emph{local section} of the admissibility sheaf
$\mathcal{A}^{(k)}_{\mathrm{RSVP}}$.

\begin{definition}[RSVP admissibility presheaf]
  \label{def:rsvp-presheaf}
  Define
  $\mathcal{A}^{(k)}_{\mathrm{RSVP}} : \mathcal{O}(X)^{op}
  \to \mathbf{Sob}_k$
  by assigning to each open set $U$:
  \[
    \mathcal{A}^{(k)}_{\mathrm{RSVP}}(U)
    = \{(\Phi_U,\mathbf{v}_U,S_U,\ell_U) \in
      \mathcal{H}^k_{\mathrm{RSVP}}(U) :
      L_U = \ell_U^2 \geq 0,\;
      \mathcal{F}_U < \infty,\;
      B_U(s_U) = 0\}.
  \]
  Restriction maps $\rho^U_V(s_U) = s_U|_V$ for $V \subseteq U$.
\end{definition}

\begin{proposition}[Restriction preserves admissibility]
  \label{prop:restrict-adm}
  \statusproven{}
  If $s_U \in \mathcal{A}(U)$ and $V \subseteq U$,
  then $s_U|_V \in \mathcal{A}(V)$.
\end{proposition}

\begin{proof}
  Local constraints are evaluated pointwise or over subregions.
  $L_U = \ell_U^2 \geq 0$ restricts to $L_V = \ell_V^2 \geq 0$.
  $\mathcal{F}_V(s_U|_V) \leq \mathcal{F}_U(s_U) < \infty$
  since the energy density is bounded below.
\end{proof}

\begin{proposition}[Presheaf functoriality]
  \label{prop:presheaf-functorial}
  \statusproven{}
  $U \mapsto \mathcal{A}(U)$ with restrictions $\rho^U_V$
  is a presheaf.
\end{proposition}

\begin{proof}
  $\rho^U_U = \mathrm{id}$ trivially.
  For $W \subseteq V \subseteq U$:
  $(\rho^V_W \circ \rho^U_V)(s_U)
  = (s_U|_V)|_W = s_U|_W = \rho^U_W(s_U)$.
\end{proof}
