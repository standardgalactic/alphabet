\chapter{Open Analytical Problems}
\label{ch:open-problems}

This chapter states the major unresolved mathematical problems
of the Holonomic Space framework.
These are not failures of the theory.
They are the research targets that determine whether the framework
advances from a well-motivated proposal to an established result.

\section{Level I: Proven variational structure}

The following are established by standard functional analysis.

\begin{theorem}[Free-energy dissipation] \statusproven{}
  Under the RSVP gradient-flow equations, the free-energy
  functional $\mathcal{F}[\Phi, \mathbf{v}, \ell, S]$
  satisfies
  \[
    \frac{d\mathcal{F}}{dt} \leq 0.
  \]
\end{theorem}

\begin{theorem}[Positivity of lamphron] \statusproven{}
  Under the reparameterisation $L = \ell^2$, the lamphron
  density satisfies $L \geq 0$ identically.
\end{theorem}

\section{Level II: Open conjectures}

\begin{conjecture}[Global RSVP existence] \statusconjectured{}
  For sufficiently smooth initial data
  $(\Phi_0, \mathbf{v}_0, \ell_0, S_0) \in H^k$,
  the fully coupled RSVP system admits a global weak solution.
\end{conjecture}

\begin{conjecture}[Admissibility reconstruction] \statusconjectured{}
  The admissibility sheaf $\mathcal{A}^{(k)}$ over
  cosmological observer domains with finite causal horizons
  possesses nontrivial obstruction classes:
  \[
    H^1(X, \mathcal{A}^{(k)}) \neq 0.
  \]
\end{conjecture}

\begin{conjecture}[Redshift emergence] \statusconjectured{}
  The eikonal limit of electromagnetic propagation through an
  RSVP medium yields an effective redshift law
  \[
    1 + z
    = \exp\!\left(\int_\gamma \mathcal{R}_{\mathrm{RSVP}}\, ds\right)
    + O(\varepsilon),
  \]
  with $\mathcal{R}_{\mathrm{RSVP}}$ determined by the local
  RSVP field configuration.
\end{conjecture}

\section{Open problems}

\begin{enumerate}[leftmargin=2em]
  \item \statusopen{} Determine the Sobolev index $k$ required
    for well-posedness of the fully coupled RSVP system.
  \item \statusopen{} Compute $H^1(X, \mathcal{A}^{(k)})$
    for explicit cosmological covers.
  \item \statusopen{} Derive the coarsening phase diagram
    $z_{\mathrm{RSVP}} = z(Pe, \Gamma, \Omega, \ldots)$
    analytically or by simulation.
  \item \statusopen{} Determine whether the redshift ansatz
    emerges from wave propagation through the RSVP medium or
    must be modified.
  \item \statusopen{} Prove or disprove the agent existence
    theorem: that for any admissible ambient field configuration,
    there exist nontrivial sections in $H^0(U, \mathcal{E})$
    for sufficiently large $U$.
\end{enumerate}
