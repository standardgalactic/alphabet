\chapter{Void Formation}
\label{ch:void-formation}

\section{Voids as entropy-dominated regions}

\begin{definition}[Void region]
  A \emph{void} in the RSVP framework is a region $V \subseteq X$
  where lamphrodyne relaxation dominates lamphron stabilisation:
  \[
    V = \{x : D(x) \gg L(x),\; |\nabla\Phi| \to 0,\; S \to S_{\max}\}.
  \]
\end{definition}

\begin{proposition}[Void growth criterion]
  \label{prop:void-growth}
  \statusproven{}
  A void expands whenever $D > (\beta/\alpha)L$, where
  $\partial_t R = \alpha D - \beta L$ is the void radius evolution.
\end{proposition}

\begin{proof}
  $\partial_t R > 0$ iff $\alpha D > \beta L$,
  i.e., $D > (\beta/\alpha)L$.
\end{proof}

\begin{proposition}[Void scaling]
  If $D \sim t^{-p}$ and $p < 1$, then $R(t) \sim t^{1-p}$.
\end{proposition}

\begin{proof}
  Integrate $\partial_t R = \alpha D \sim \alpha t^{-p}$:
  $R(t) \sim \alpha t^{1-p}/(1-p)$ for $p < 1$.
\end{proof}

\begin{ObsConseq}[Void statistics]
  \textbf{Mechanism:} Lamphrodyne-dominated entropy descent.

  \textbf{Prediction:} Void size distribution follows
  $N(R) \propto R^{-(1-p)^{-1}}$ for appropriate exponent $p$.

  \textbf{Comparison:} SDSS, DES void catalogues.

  \textbf{Status:} \statusopen{}
  Requires determination of $p$ from RSVP field equations.
\end{ObsConseq}
