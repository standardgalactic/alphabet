\chapter{Filament Formation}
\label{ch:filament-formation}

\section{Filaments as topological defects}

\begin{definition}[Filament]
  A \emph{filament} in the RSVP framework is a region where
  lamphron stabilisation dominates:
  \[
    F = \{x : L \gg D,\; |\nabla\Phi| \text{ large},\;
      \nabla\times\mathbf{v} \neq 0\}.
  \]
\end{definition}

\begin{definition}[RSVP defect set]
  \[
    \mathcal{D} = \{x : \nabla\Phi \text{ singular or discontinuous}\}.
  \]
\end{definition}

\begin{proposition}[Persistent structures are local free-energy minima]
  \statusproven{}
  A defect at $x \in \mathcal{D}$ is stable if perturbing it
  increases $\mathcal{F}$: $\delta^2\mathcal{F} > 0$.
\end{proposition}

\begin{theorem}[Filament linear stability]
  \label{thm:filament-stable}
  \statusproven{}
  Let $\Gamma$ be a filament.
  Perturb: $\Gamma \to \Gamma + \varepsilon h$.
  If the second variation satisfies
  \[
    \delta^2\mathcal{F}
    = \int_\Gamma \left(|\nabla h|^2 + Kh^2\right)ds > 0
  \]
  (i.e., $K > 0$), the filament is linearly stable.
\end{theorem}

\section{Cosmic web topology}

\begin{definition}[Cosmic web complex]
  Let $K$ be the simplicial complex with:
  vertices = cluster nodes,
  edges = filaments,
  faces = walls.
  The Euler characteristic is $\chi(K) = V - E + F$.
\end{definition}

\begin{proposition}[Topological changes require defect events]
  \statusproven{}
  Continuous deformations preserve homotopy type.
  Only creation or annihilation of defects changes
  the topology of $K$.
\end{proposition}
