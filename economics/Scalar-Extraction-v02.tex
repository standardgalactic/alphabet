\begin{abstract}
Contemporary social platforms instantiate a dynamical system that concentrates visibility potential $\Phi$, suppresses agency vectors $\vvec$, and amplifies informational entropy $\Entropy$. 
We formalize this extraction as a field theory in which $\nabla\Phi \cdot \vvec < 0$ and $\nabla \Entropy \cdot \vvec > 0$, forcing visibility to oppose agency and disorder to grow with effort. 
We derive necessary and sufficient conditions for extraction, introduce Lyapunov stability guarantees for non-extractive regimes, treat visibility markets as monopsonistic auctions over attention labor, and provide adversarial thresholds for Sybil, entropy, and collusion attacks. 
We propose an enforceable constitutional architecture for non-extractive networks using scalar caps, cooperative credit decay, entropy damping, and reservoir-based visibility recirculation. 
The result is a falsifiable, measurable, and implementable theory of sociotechnical extraction and its reversal.
\end{abstract}

\noindent\textbf{Keywords:} attention economics, entropy regulation, network field theory, auction markets, digital governance, extractive platforms, cooperative ranking.

\section{Axioms and Definitions}

\begin{definition}[Extractive Platform]
A platform is extractive if:
\begin{enumerate}
    \item $\Phi$ is artificially scarce but purchasable,
    \item $\vvec$ cannot increase $\Phi$ without payment,
    \item $\vvec$ increases $\Entropy$ rather than reducing it,
    \item platform profit grows with $\Entropy$,
    \item users cannot resolve visibility uncertainty without payment.
\end{enumerate}
\end{definition}

\begin{axiom}[Visibility Conservation Violation]
In non-extractive communication systems, visibility is conserved:
$\sum_x \Phi_x = C$.
Extractive platforms violate conservation via:
\[
\sum_x \Phi_x < C \quad \text{and} \quad \frac{\partial}{\partial t} C_{\text{available}} < 0
\]
\end{axiom}

\begin{axiom}[Agency Opposition]
Platform gradients oppose agency:
\[
\nabla \Phi \cdot \vvec < 0
\]
\end{axiom}

\begin{axiom}[Entropy Alignment]
User actions increase entropy in the system:
\[
\nabla \Entropy \cdot \vvec > 0
\]
\end{axiom}

\begin{lemma}[Extraction Criterion]
A system extracts iff:
\[
\mathbb{E}[\nabla \Phi \cdot \vvec] < 0 \quad \wedge \quad \mathbb{E}[\nabla\Entropy \cdot \vvec] > 0
\]
\end{lemma}

\begin{remark}
This corresponds to the phenomenology of ``work harder, reach less''.
\end{remark}

\begin{example}
A creator posts organically, reaches 50 users, pays to boost to 500, then returns to 20.  
Effort increased, entropy increased, reach fell: extraction confirmed.
\end{example}

\begin{definition}[Visibility Constitutional Cap]
A platform respects constitutional influence limits if there exists a constant $\Phi_{\max}$ such that for all users $x$:
\[

and cumulative systemic visibility never exceeds:
\[

\begin{proposition}[No Infinite Amplification]
If visibility gains are bounded by:
\[
\Phi_x(t+1) = \min(\Phi_{\max},\ \Phi_x(t) + \Delta \Phi_x)

\end{proposition}

\begin{proof}
Trivial from monotone bounded convergence: $\Phi_x(t)$ is increasing but bounded above; hence $\lim_{t \to \infty} \Phi_x(t) \le \Phi_{\max}$.
\end{proof}

import numpy as np
import networkx as nx
from scipy.stats import entropy

class PlatformField:
    def __init__(self, n_agents=1000, rho=0.97, phi_max=10.0):
        self.n = n_agents
        self.rho = rho
        self.phi_max = phi_max
        self.Phi = np.random.rand(n_agents)
        self.C = np.zeros(n_agents)
        self.actions = np.random.randint(0, 20, size=n_agents)
        self.T = np.zeros((20,20))

    def update_credit(self, delta):
        self.C = self.rho * self.C + delta
        return self.C

    def update_visibility(self, extraction_strength=0.1):
        grad = np.gradient(self.Phi)
        self.Phi = self.Phi + extraction_strength * grad
        self.Phi = np.clip(self.Phi, 0, self.phi_max)
        return self.Phi

    def update_action_transitions(self):
        for i in range(len(self.actions)-1):
            a, b = self.actions[i], self.actions[i+1]
            self.T[a,b] += 1
        self.T /= (self.T.sum(axis=1, keepdims=True) + 1e-6)
        return self.T

    def gini_visibility(self):
        diff = np.abs(self.Phi[:,None] - self.Phi[None,:])
        return diff.sum() / (2 * self.n * self.Phi.sum())

    def entropy_attention(self):
        p = self.Phi / (self.Phi.sum() + 1e-9)
        return entropy(p)

    def action_rank(self):
        return np.linalg.matrix_rank(self.update_action_transitions())

    def inject_noise(self, scale=0.2):
        self.Phi += np.random.randn(self.n) * scale
        self.Phi = np.clip(self.Phi, 0, self.phi_max)

    def step(self, extraction=0.1, noise=0.0):
        self.update_visibility(extraction)
        self.update_credit(np.random.randn(self.n) * 0.05)
        if noise: self.inject_noise(noise)
        return {
            "gini": self.gini_visibility(),
            "entropy": self.entropy_attention(),
            "rank": self.action_rank(),
        }

def run_trials(T=200):
    pf = PlatformField()
    history = []
    for t in range(T):
        stats = pf.step(extraction=0.15, noise=0.03)
        history.append(stats)
    return history

if __name__ == "__main__":
    h = run_trials()
    print(h[-1])
