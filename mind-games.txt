    kx, ky, kz = np.meshgrid(*[np.fft.fftfreq(N) * N * dx for _ in range(3)], indexing='ij')
    
    # Fourier transforms
    phi_k = jax.numpy.fft.fftn(phi)
    v_k = jax.numpy.fft.ifftn(v_k)

    # Compute curl_v_k
    curl_v_k = np.array([
        v_k[2] * 1j * ky - v_k[1] * 1j * kz,
        v_k[0] * 1j * kz - v_k[2] * 1j * kx,
        v_k[1] * 1j * kx - v_k[0] * 1j * ky
    ])

    # Compute curl_curl_v_k
    curl_curl_v_k = np.array([
        curl_v_k[2] * 1j * ky - curl_v_k[1] * 1j * kz,
        curl_v_k[0] * 1j * kz - curl_v_k[2] * 1j * kx,
        curl_v_k[1] * 1j * kx - curl_v_k[0] * 1j * ky
    ])

    # Update v_k with vorticity suppression and curl-curl dynamics
    v_k -= epsilon * jax.numpy.fft.ifftn(curl_curl_v_k) / N**3
