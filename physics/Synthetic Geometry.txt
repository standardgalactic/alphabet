import numpy as np
import matplotlib.pyplot as plt

# Simulation parameters
N = 100  # Lattice size (N x N)
dt = 0.01  # Time step
kappa = 0.1  # Coupling constant for vector divergence
alpha = 0.05  # Diffusion coefficient for entropy smoothing
num_steps = 100  # Number of time steps

# Initialize fields
S = np.random.normal(0, 0.1, (N, N))  # Entropy field (random initial condition)
Phi = np.sin(0.1 * np.indices((N, N))[0])  # Scalar field (sinusoidal profile)
vx = np.cos(0.1 * np.indices((N, N))[1])  # Vector field x-component
vy = np.sin(0.1 * np.indices((N, N))[0])  # Vector field y-component

# Compute divergence of vector field
def compute_divergence(vx, vy, dx=1.0):
    div_v = np.gradient(vx, dx, axis=1) + np.gradient(vy, dx, axis=0)
    return div_v

# Compute Laplacian of entropy field
def compute_laplacian(S, dx=1.0):
    laplacian = (np.roll(S, 1, axis=0) + np.roll(S, -1, axis=0) +
                 np.roll(S, 1, axis=1) + np.roll(S, -1, axis=1) - 4 * S) / dx**2
    return laplacian

# Update entropy field
def update_S(S, Phi, vx, vy, dt, kappa, alpha):
    div_v = compute_divergence(vx, vy)
    laplacian_S = compute_laplacian(S)
    dS = kappa * div_v - alpha * laplacian_S  # Entropy update rule
    return S + dt * dS

# Simulation loop
S_history = [S.copy()]
for step in range(num_steps):
    S = update_S(S, Phi, vx, vy, dt, kappa, alpha)
    S_history.append(S.copy())

plt.title('Entropy Field at t = {:.2f}'.format(num_steps * dt))

import numpy as np
from scipy.interpolate import UnivariateSpline
import matplotlib.pyplot as plt
from cobaya.run import run
from cobaya.model import Model
from cobaya.likelihoods.base_classes import BAO, SN, CMBLensing
from getdist import plots
import os

# Stage 1: Extract Λ_eff and fσ8 from simulation traces
def extract_lambda_eff_and_fsigma8(s_history, vx_history, vy_history, vz_history, dt, dx=1.0):
    def compute_divergence(vx, vy, vz, dx):
        div_v = (np.roll(vx, -1, axis=0) - np.roll(vx, 1, axis=0) +
                 np.roll(vy, -1, axis=1) - np.roll(vy, 1, axis=1) +
                 np.roll(vz, -1, axis=2) - np.roll(vz, 1, axis=2)) / (2 * dx)
        return div_v
    
    t_steps = len(s_history)
    Lambda_eff = []
    fsigma8 = []
    sigma8_0 = 0.811  # Fiducial σ8 at z=0 (Planck 2018)
    for t in range(1, t_steps-1):
        div_v = compute_divergence(vx_history[t], vy_history[t], vz_history[t], dx)
        d_div_v_dt = (compute_divergence(vx_history[t+1], vy_history[t+1], vz_history[t+1], dx) -
                      compute_divergence(vx_history[t-1], vy_history[t-1], vz_history[t-1], dx)) / (2 * dt)
        dS_dt = (s_history[t+1] - s_history[t-1]) / (2 * dt)
        Lambda_eff_t = np.mean(d_div_v_dt - dS_dt)
        f_t = np.mean(np.abs(div_v)) / 3  # Proxy for growth rate f ≈ (∇·v)/3
        fsigma8_t = f_t * sigma8_0  # Simplified, assumes linear growth
        Lambda_eff.append(Lambda_eff_t)
        fsigma8.append(fsigma8_t)
    return np.array(Lambda_eff), np.array(fsigma8)

# Stage 2: Map time to redshift
def time_to_redshift(t, vx_history, vy_history, vz_history, dx=1.0):
    H = []
    for t_idx in range(len(vx_history)):
        div_v = compute_divergence(vx_history[t_idx], vy_history[t_idx], vz_history[t_idx], dx)
        H_t = np.mean(div_v) / 3
        H.append(H_t)
    H = np.array(H)
    z = np.exp(np.cumsum(H[:-1] * dt)) - 1
    return z

# Stage 3: Cobaya model setup
def setup_cobaya_model(z, Lambda_eff, fsigma8):
    Lambda_spline = UnivariateSpline(z, Lambda_eff, s=0.1, k=3, ext='const')
    fsigma8_spline = UnivariateSpline(z, fsigma8, s=0.05, k=3, ext='const')
    
    def theory_model(z, Lambda0, gamma, H0, omegam, omegab):
        Lambda_eff_z = Lambda0 * Lambda_spline(z) / Lambda_spline(0) * (1 + z)**(-gamma)
        fsigma8_z = fsigma8_spline(z)
        # Simplified comoving distance for BAO
        DM = np.cumsum(1 / (H0 * np.sqrt(omegam * (1+z)**3 + (1-omegam) * Lambda_eff_z / Lambda0))) * dt
        return {'Lambda_eff': Lambda_eff_z, 'fsigma8': fsigma8_z, 'DM': DM}
    
    info = {
        'params': {
            'Lambda0': {'prior': {'min': 0.1, 'max': 1.0}, 'ref': 0.7, 'proposal': 0.05},
            'gamma': {'prior': {'min': -1.0, 'max': 1.0}, 'ref': 0.1, 'proposal': 0.1},
            'H0': {'prior': {'dist': 'norm', 'loc': 73.04, 'scale': 1.04}, 'ref': 73.04, 'proposal': 0.5},
            'omegam': {'prior': {'min': 0.1, 'max': 0.5}, 'ref': 0.3, 'proposal': 0.01},
            'omegab': {'prior': {'min': 0.03, 'max': 0.07}, 'ref': 0.05, 'proposal': 0.005}
        },
        'likelihood': {
            'desi_bao_2024': {
                'type': 'BAO',
                'data_file': 'desi_bao_2024.dat',
                'z': [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0],
                'data_type': 'DM_over_rd'
            },
            'pantheon_plus': {
                'type': 'SN',
                'data_file': 'pantheon_plus.dat',
                'z': np.linspace(0.001, 2.26, 100),
                'data_type': 'magnitude'
            },
            'planck_2018_lensing': {
                'type': 'CMBLensing',
                'data_file': 'planck_2018_lensing.dat',
                'z': [0, 1100],
                'data_type': 'Cl'
            },
            'fsigma8_likelihood': {
                'type': 'Likelihood',
                'data': {'z': [0.38, 0.51, 0.61], 'fsigma8': [0.47, 0.45, 0.43], 'sigma': [0.03, 0.03, 0.03]},
                'function': lambda z, fsigma8: np.sum(-0.5 * ((fsigma8 - fsigma8_spline(z)) / 0.03)**2)
            }
        },
        'theory': {
            'cosmology': {
                'function': theory_model,
                'params': ['Lambda0', 'gamma', 'H0', 'omegam', 'omegab']
            }
        },
        'sampler': {
            'mcmc': {
                'Rminus1_stop': 0.01,
                'max_tries': 1000
            }
        }
    }
    return info, Lambda_spline, fsigma8_spline

# Stage 4: Run Cobaya and plot results
def run_and_plot():
    S_history = np.load('S_history.npy')
    vx_history = np.load('vx_history.npy')
    vy_history = np.load('vy_history.npy')
    vz_history = np.load('vz_history.npy')
    dt = 0.01
    dx = 1.0
    
    Lambda_eff, fsigma8 = extract_lambda_eff_and_fsigma8(S_history, vx_history, vy_history, vz_history, dt, dx)
    t = np.arange(1, len(S_history)-1) * dt
    z = time_to_redshift(t, vx_history, vy_history, vz_history, dx)
    
    info, Lambda_spline, fsigma8_spline = setup_cobaya_model(z, Lambda_eff, fsigma8)
    updated_info, sampler = run(info)
    
    z_fine = np.linspace(0, 2, 100)
    Lambda_eff_z = Lambda_spline(z_fine)
    w_z = -1 + Lambda_eff_z / (3 * updated_info['params']['Lambda0']['ref'])
    fsigma8_z = fsigma8_spline(z_fine)
    
    plt.figure(figsize=(15, 5))
    plt.subplot(1, 3, 1)
    plt.plot(z_fine, Lambda_eff_z, label='$\Lambda_{\text{eff}}(z)$')
    plt.axhline(0.7, color='k', linestyle='--', label='$\Lambda$CDM')
    plt.xlabel('Redshift ($z$)')
    plt.ylabel('$\Lambda_{\text{eff}}(z)$')
    plt.legend()
    plt.title('Effective Cosmological Term')
    
    plt.subplot(1, 3, 2)
    plt.plot(z_fine, w_z, label='$w(z)$')
    plt.axhline(-1, color='k', linestyle='--', label='$\Lambda$CDM')
    plt.xlabel('Redshift ($z$)')
    plt.ylabel('$w(z)$')
    plt.legend()
    plt.title('Dark Energy Equation of State')
    
    plt.subplot(1, 3, 3)
    plt.plot(z_fine, fsigma8_z, label='$f\sigma_8(z)$')
    plt.scatter([0.38, 0.51, 0.61], [0.47, 0.45, 0.43], color='r', label='Data')
    plt.xlabel('Redshift ($z$)')
    plt.ylabel('$f\sigma_8(z)$')
    plt.legend()
    plt.title('Structure Growth')
    plt.tight_layout()
    plt.savefig('cosmological_fit_extended.png')
    plt.show()
    
    g = plots.get_subplot_plotter()
    g.triangle_plot(sampler.products()['sample'], ['Lambda0', 'gamma', 'H0', 'omegam', 'omegab'])
    plt.savefig('posterior_corner.png')
    plt.show()
    
    np.save('posteriors_extended.npy', sampler.products()['sample'])

if __name__ == '__main__':
    run_and_plot()

### 2. Symbolic Tensor Derivation

The stress-energy tensor \(T_{AB}\) is derived from the GU-RSVP action:

\[
S = \int d^{14}x \sqrt{-G} \left[ \alpha \mathcal{R}_{\text{eff}}(G_{AB}, v^C, S) + \beta \nabla_A S \nabla^A \Phi - \gamma (\nabla_A v_B)(\nabla^A v^B) - V(\Phi, S) \right]
\]

Variation w.r.t. \(G_{AB}\) yields:

\[
T_{AB} = \beta \left( \nabla_A S \nabla_B \Phi + \nabla_B S \nabla_A \Phi - G_{AB} \nabla_C S \nabla^C \Phi \right) + \gamma \left( \nabla_A v_C \nabla_B v^C + \nabla_B v_C \nabla_A v^C - G_{AB} \nabla_D v_E \nabla^D v^E \right) - G_{AB} V(\Phi, S)
\]

Christoffel symbols \(\Gamma^C_{AB}\) are computed for the effective metric \(G_{AB}\).

\section{Action and Variation}
The GU-RSVP action is:
\[
S = \int d^{14}x \sqrt{-G} \left[ \alpha \mathcal{R}_{\text{eff}}(G_{AB}, v^C, S) + \beta \nabla_A S \nabla^A \Phi - \gamma (\nabla_A v_B)(\nabla^A v^B) - V(\Phi, S) \right]

where \(\mathcal{R}_{\text{eff}} = R + \lambda_1 \left(\frac{\delta S}{\delta \Phi}\right)^2 + \lambda_2 v^A v^B R_{AB}\), and \(V(\Phi, S) = \mu \Phi^2 + \nu S^2 + \xi \Phi^2 S^2\).

Varying w.r.t. \(G_{AB}\):
\[
\frac{\delta S}{\delta G^{AB}} = \alpha \left( R_{AB} - \frac{1}{2} G_{AB} R \right) - T_{AB} = 0
\]
The stress-energy tensor is:
\[
T_{AB} = \beta \left( \nabla_A S \nabla_B \Phi + \nabla_B S \nabla_A \Phi - G_{AB} \nabla_C S \nabla^C \Phi \right) + \gamma \left( \nabla_A v_C \nabla_B v^C + \nabla_B v_C \nabla_A v^C - G_{AB} \nabla_D v_E \nabla^D v^E \right) - G_{AB} V(\Phi, S)
\]

\section{Christoffel Symbols}
For the effective metric \(G_{AB}\):
\[
\Gamma^C_{AB} = \frac{1}{2} G^{CD} \left( \partial_A G_{DB} + \partial_B G_{DA} - \partial_D G_{AB} \right)
\]
In a 4D projection (e.g., \(G_{\mu\nu}\)):
\[
\Gamma^\rho_{\mu\nu} = \frac{1}{2} G^{\rho\sigma} \left( \partial_\mu G_{\sigma\nu} + \partial_\nu G_{\sigma\mu} - \partial_\sigma G_{\mu\nu} \right)
\]
These are computed numerically in lattice simulations using finite differences.
