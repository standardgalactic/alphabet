\section{Proof of Theorem 1: Global Weak Solutions}
\label{app:weak}

We prove existence via Galerkin approximation.

Let $\{w_k\}_{k=1}^\infty$ be the eigenfunctions of $-\Delta$ with Neumann conditions, orthonormal in $L^2(\Omega)$. Define the finite-dimensional projection
\[
\Phi^n = \sum_{k=1}^n \phi_k(t) w_k(x), \quad \mathbf{v}^n = \sum_{k=1}^n \mathbf{v}_k(t) w_k(x), \quad S^n = \sum_{k=1}^_element s_k(t) w_k(x).
\]

The Galerkin system is
\begin{align}
\frac{d\phi_j}{dt} &= -\sum_k \phi_k \mathbf{v}_k \cdot \nabla w_j + \kappa_\Phi \lambda_j \phi_j, \\
\frac{d\mathbf{v}_j}{dt} &= -\sum_k (\mathbf{v}_k \cdot \nabla) \mathbf{v}_j - \sum_k \phi_k \nabla w_j + \kappa_v \lambda_j \mathbf{v}_j, \\
\frac{ds_j}{dt} &= \sigma^n_j + \kappa_S \lambda_j s_j - \lambda s_j,
\end{align}
where $\sigma^n_j = \alpha |\nabla \Phi^n \cdot \mathbf{v}^n| w_j$.

\textbf{Energy Estimate}: Multiply by $\phi_j, \mathbf{v}_j, s_j$, sum over $j$:
\[
\frac{1}{2} \frac{d}{dt} \left( \|\Phi^n\|^2 + \|\mathbf{v}^n\|^2 + \|S^n\|^2 \right) + \kappa_\Phi \|\nabla \Phi^n\|^2 + \kappa_v \|\nabla \mathbf{v}^n\|^2 + \kappa_S \|\nabla S^n\|^2 + \lambda \|S^n\|^2 \le C(1 + \|\Phi^n\| + \|\mathbf{v}^n\|^2).
\]
Gronwall yields uniform bounds in $L^\infty(0,T; H^1) \cap L^2(0,T; H^2)$.

\textbf{Compactness}: Aubin--Lions lemma gives strong convergence in $L^2(0,T; L^2)$.

\textbf{Passage to Limit}: Weak-* convergence in $L^\infty(0,T; H^1)$ and strong in $L^2(0,T; L^2)$ allow passage to weak form.

\section{Proof of Entropy Bound (Lemma 2)}
\label{app:entropy}

Integrate \eqref{eq:S}:
\[
\frac{d}{dt} \int S = \int \sigma - \lambda \int S.
\]
Since $\sigma \le \alpha \|\nabla \Phi\|_\infty \|\mathbf{v}\|_\infty \le C(1 + \|\Phi\|_{H^1} + \|\mathbf{v}\|_{H^1})$ and energy bounds are uniform, $\int \sigma \le K$. Then
\[
\int S(t) \le e^{-\lambda t} \int S_0 + \frac{K}{\lambda} (1 - e^{-\lambda t}) \le B_S.
\]

\section{Proof of Entropic Redshift Monotonicity}
\label{app:redshift}

From \eqref{eq:z},
\[
\frac{d}{dt} \log(1 + z_{\mathrm{RSVP}}) = \alpha_S \frac{1}{|\Omega|} \int_\Omega \partial_t S\,dx = \alpha_S \left( \frac{1}{|\Omega|} \int \sigma - \lambda \langle S \rangle \right).
\]
Since $\sigma \ge 0$ and $\lambda \langle S \rangle \le \lambda B_S/|\Omega|$, the integral is bounded below by $-\lambda B_S/|\Omega|$, but dominant positive contribution from $\sigma$ ensures monotonicity on average.

\section{Linear Stability Analysis}
\label{app:stability}

Linearize around $(\Phi_*,\mathbf{0},S_*)$:
\[
\partial_t \delta \Phi = -\Phi_* \nabla \cdot \delta \mathbf{v} + \kappa_\Phi \Delta \delta \Phi,
\]
\[
\partial_t \delta \mathbf{v} = -\nabla \delta \Phi + \kappa_v \Delta \delta \mathbf{v},
\]
\[
\partial_t \delta S = \alpha |\nabla \Phi_* \cdot \delta \mathbf{v}| + \kappa_S \Delta \delta S - \lambda \delta S.
\]

Fourier mode $e^{\lambda t + i\mathbf{k}\cdot\mathbf{x}}$ yields characteristic equation
\[
\det\begin{pmatrix}
\lambda + \kappa_\Phi k^2 & i k_x \Phi_* & i k_y \Phi_* & i k_z \Phi_* \\
i k_x & \lambda + \kappa_v k^2 & 0 & 0 \\
i k_y & 0 & \lambda + \kappa_v k^2 & 0 \\
i k_z & 0 & 0 & \lambda + \kappa_v k^2
\end{pmatrix} = 0.
\]

For $k \ne 0$, all eigenvalues have negative real part by Gershgorin.
