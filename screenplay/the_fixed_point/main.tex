% ── Interlude environment ────────────────────────────────────
% No named characters. Object-focused. Slightly lighter type via spacing.
\newenvironment{interlude}[1]{%
  \clearpage
  \thispagestyle{plain}
  \begin{center}
    {\small\scshape object}\\[0.8ex]
    {\normalsize\itshape #1}\\[2.0ex]
    {\color{ruledline}\rule{0.52\textwidth}{0.25pt}}
  \end{center}
  \vspace{1.8ex}
  \begin{adjustwidth}{1.4em}{1.4em}
  \setstretch{1.16}%
}{%
  \end{adjustwidth}
  \vspace{2.0ex}
  \begin{center}
    {\color{ruledline}\rule{0.52\textwidth}{0.25pt}}
  \end{center}
  \clearpage
}
